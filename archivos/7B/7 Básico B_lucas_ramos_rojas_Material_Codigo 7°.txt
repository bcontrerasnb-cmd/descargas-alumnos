    <script>
        // FUNCIÓN: CAMBIAR ENTRE VISTAS SIN RECARGAR LA PÁGINA
        // Recibe como parámetro el ID de la sección que queremos mostrar.
        function cambiarVista(idVistaDestino) {
            // Paso 1: Buscamos todas las secciones en el documento y las ocultamos ('none').
            document.getElementById('vista-inicio').style.display = 'none';
            document.getElementById('vista-publicaciones').style.display = 'none';
            document.getElementById('vista-crear').style.display = 'none';
            document.getElementById('vista-editar').style.display = 'none';
            
            // Paso 2: Buscamos la sección que pidió el usuario y la mostramos en pantalla ('block').
            document.getElementById(idVistaDestino).style.display = 'block';
        }

        // FUNCIÓN: SABER QUÉ TIPO DE IMAGEN SUBIÓ EL USUARIO
        function obtenerImagen(idArchivo, idUrl, idActual = null) {
            // Buscamos si hay un archivo seleccionado en el PC y si escribieron un texto en la URL.
            let archivo = document.getElementById(idArchivo).files[0];
            let url = document.getElementById(idUrl).value;
            
            if (archivo) {
                // Si subió un archivo, le pedimos al navegador que invente un link temporal para poder verlo.
                return URL.createObjectURL(archivo); 
            } else if (url !== "") {
                // Si pegó una URL de internet, devolvemos ese mismo link.
                return url; 
            } else if (idActual) {
                // Si estamos editando y no subió nada nuevo, mantenemos la imagen vieja oculta en el HTML.
                return document.getElementById(idActual).value; 
            }
            // Si dejó todo vacío, devolvemos un texto vacío.
            return ""; 
        }

        // FUNCIÓN: CREAR UNA NUEVA PUBLICACIÓN (La 'C' de CRUD)
        function crearPost() {
            // 1. Rescatamos lo que el usuario escribió en las cajas de texto (.value).
            let titulo = document.getElementById("tit").value;
            let texto = document.getElementById("txt").value;
            
            // 2. Llamamos a nuestra función ayudante para ver qué imagen eligió.
            let imagenFinal = obtenerImagen("imgArchivo", "imgUrl");

            // 3. Validamos: si el título Y el texto están vacíos, lanzamos una alerta y cortamos la función (return).
            if(titulo === "" && texto === "") {
                alert("¡Debes escribir un título o texto antes de publicar!");
                return; // Este return frena la ejecución aquí.
            }

            // 4. Buscamos el div "muro" y le sumamos (+=) el código HTML con los datos inyectados.
            // Usamos las comillas invertidas (`) para poder escribir HTML de múltiples líneas y variables.
            document.getElementById("muro").innerHTML += `
                <div class="tarjeta-post">
                    <h3 class="t-post">${titulo}</h3>
                    <img src="${imagenFinal}" class="img-post">
                    <p class="p-post">${texto}</p>
                    <div class="caja-botones">
                        <!-- El botón 'this' envía toda la información de sí mismo a la función -->
                        <button class="btn btn-secundario" onclick="abrirEditar(this)">Editar</button>
                        <button class="btn btn-peligro" onclick="borrarPost(this)">Eliminar</button>
                    </div>
                </div>
            `;
            
            // 5. Limpiamos las cajas del formulario para que queden vacías para el próximo post.
            document.getElementById("tit").value = "";
            document.getElementById("txt").value = "";
            document.getElementById("imgUrl").value = "";
            document.getElementById("imgArchivo").value = "";

            // 6. Finalmente, nos cambiamos de pantalla para ver el muro actualizado.
            cambiarVista('vista-publicaciones');
        }

        // FUNCIÓN: ELIMINAR UNA PUBLICACIÓN (La 'D' de CRUD)
        function borrarPost(botonApretado) {
            // Lanzamos una alerta que permite elegir Aceptar (true) o Cancelar (false).
            let confirmacion = confirm("¿Seguro que quieres eliminar esta publicación?");
            
            // Si el usuario dijo que Sí...
            if (confirmacion === true) {
                // Buscamos al "abuelo" del botón (El botón está dentro de una caja, y la caja está dentro de la tarjeta).
                // y usamos .remove() para destruir toda la tarjeta HTML de la pantalla.
                botonApretado.parentElement.parentElement.remove(); 
            }
        }

        // FUNCIÓN: PREPARAR LA EDICIÓN (La 'U' de CRUD - Parte 1)
        // Creamos una variable global para que el computador "recuerde" qué tarjeta estamos editando.
        let postAEditar = null; 

        function abrirEditar(botonApretado) {
            // Guardamos la tarjeta entera (el "abuelo" del botón) en la memoria del computador.
            postAEditar = botonApretado.parentElement.parentElement; 
            
            // Extraemos el texto visible (.innerText) del HTML de la tarjeta y lo ponemos en los inputs del formulario (.value).
            // Usamos querySelector para buscar la clase específica solo dentro de esa tarjeta.
            document.getElementById("edit-tit").value = postAEditar.querySelector('.t-post').innerText;
            document.getElementById("edit-txt").value = postAEditar.querySelector('.p-post').innerText;
            
            // Guardamos el link de la imagen vieja en una caja invisible (hidden) por si no quieren cambiarla.
            document.getElementById("edit-imgActual").value = postAEditar.querySelector('.img-post').src;
            
            // Vaciamos las cajas de subida de imagen nueva por defecto.
            document.getElementById("edit-imgArchivo").value = "";
            document.getElementById("edit-imgUrl").value = "";
            
            // Mostramos el formulario de edición en pantalla.
            cambiarVista('vista-editar');
        }

        // FUNCIÓN: GUARDAR LOS CAMBIOS DE EDICIÓN (La 'U' de CRUD - Parte 2)
        function guardarEdicion() {
            // 1. Rescatamos los nuevos textos que el usuario modificó en el formulario.
            let nuevoTit = document.getElementById("edit-tit").value;
            let nuevoTxt = document.getElementById("edit-txt").value;
            
            // 2. Evaluamos si subió una foto nueva o nos quedamos con la foto vieja oculta.
            let nuevaImg = obtenerImagen("edit-imgArchivo", "edit-imgUrl", "edit-imgActual");
            
            // 3. Volvemos a la tarjeta que teníamos guardada en la memoria (postAEditar).
            // Y reemplazamos (.innerText y .src) los datos antiguos por los nuevos.
            postAEditar.querySelector('.t-post').innerText = nuevoTit;
            postAEditar.querySelector('.p-post').innerText = nuevoTxt;
            postAEditar.querySelector('.img-post').src = nuevaImg;
            
            // 4. Volvemos al muro para ver el resultado.
            cambiarVista('vista-publicaciones');
        }
    </script>
