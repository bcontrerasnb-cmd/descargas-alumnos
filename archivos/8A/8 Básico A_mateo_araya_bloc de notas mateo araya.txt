        let posicionEnLaLista = null;

        // CAMBIAR DE PANTALLA
        function cambiarVista(pantallaDestino) {
            const pantallas = [
                "vista-inicio",
                "vista-publicaciones",
                "vista-crear",
                "vista-editar",
                "informacion"
            ];

            pantallas.forEach(id => {
                document.getElementById(id).style.display = "none";
            });

            document.getElementById(pantallaDestino).style.display = "block";

            if (pantallaDestino === "vista-publicaciones") {
                mostrarPosts();
            }

            window.scrollTo(0, 0);
        }

        // MOSTRAR PUBLICACIONES
        function mostrarPosts() {
            let cajaMuro = document.getElementById("muro");
            cajaMuro.innerHTML = "";

            if (miListaDePublicaciones.length === 0) {
                cajaMuro.innerHTML =
                    '<div class="sin-posts"><h3>NO HAY PUBLICACIONES</h3><p>¡Sé el primero en compartir información!</p></div>';
                return;
            }

            for (let i = 0; i < miListaDePublicaciones.length; i++) {
                let post = miListaDePublicaciones[i];

                let tarjeta = document.createElement("article");
                tarjeta.className = "tarjeta-post";

                let categoria = document.createElement("div");
                categoria.className = "categoria-post";
                categoria.textContent = "// " + post.categoria;

                let titulo = document.createElement("h3");
                titulo.className = "t-post";
                titulo.textContent = post.titulo;

                tarjeta.appendChild(categoria);
                tarjeta.appendChild(titulo);

                if (post.imagen) {
                    let imagen = document.createElement("img");
                    imagen.src = post.imagen;
                    imagen.className = "img-post";
                    imagen.alt = post.titulo;
                    tarjeta.appendChild(imagen);
                }

                let texto = document.createElement("p");
                texto.className = "p-post";
                texto.textContent = post.texto;
                tarjeta.appendChild(texto);

                let botones = document.createElement("div");
                botones.className = "caja-botones";

                let editar = document.createElement("button");
                editar.className = "btn btn-secundario";
                editar.textContent = "EDITAR";
                editar.onclick = () => abrirEditar(i);

                let eliminar = document.createElement("button");
                eliminar.className = "btn btn-peligro";
                eliminar.textContent = "ELIMINAR";
                eliminar.onclick = () => borrarPost(i);

                botones.appendChild(editar);
                botones.appendChild(eliminar);
                tarjeta.appendChild(botones);

                cajaMuro.appendChild(tarjeta);
            }
        }

        // OBTENER IMAGEN
        function obtenerImagen(archivoId, linkId, imagenAnterior = "") {
            let archivo = document.getElementById(archivoId).files[0];
            let enlace = document.getElementById(linkId).value.trim();

            if (archivo) {
                return new Promise((resolve, reject) => {
                    let lector = new FileReader();
                    lector.onload = () => resolve(lector.result);
                    lector.onerror = () => reject(new Error("No se pudo leer la imagen"));
                    lector.readAsDataURL(archivo);
                });
            }

            if (enlace !== "") {
                return Promise.resolve(enlace);
            }

            return Promise.resolve(imagenAnterior);
        }

        // CREAR PUBLICACIÓN
        async function crearPost() {
            let titulo = document.getElementById("tit").value.trim();
            let texto = document.getElementById("txt").value.trim();
            let categoria = document.getElementById("categoria").value;

            if (titulo === "" || texto === "") {
                alert("¡Debes escribir un título y un contenido!");
                return;
            }

            try {
                let imagen = await obtenerImagen("imgArchivo", "imgUrl");

                miListaDePublicaciones.push({
                    titulo: titulo,
                    categoria: categoria,
                    imagen: imagen,
                    texto: texto
                });

                document.getElementById("tit").value = "";
                document.getElementById("txt").value = "";
                document.getElementById("imgUrl").value = "";
                document.getElementById("imgArchivo").value = "";
                document.getElementById("categoria").selectedIndex = 0;

                cambiarVista("vista-publicaciones");
            } catch (error) {
                alert("No se pudo cargar la imagen.");
            }
        }

        // ELIMINAR PUBLICACIÓN
        function borrarPost(numero) {
            if (confirm("¿Seguro que quieres eliminar esta publicación?")) {
                miListaDePublicaciones.splice(numero, 1);
                mostrarPosts();
            }
        }

        // ABRIR EDITOR
        function abrirEditar(numero) {
            posicionEnLaLista = numero;

            let post = miListaDePublicaciones[numero];

            document.getElementById("edit-tit").value = post.titulo;
            document.getElementById("edit-categoria").value = post.categoria;
            document.getElementById("edit-txt").value = post.texto;
            document.getElementById("edit-imgActual").value = post.imagen;
            document.getElementById("edit-imgUrl").value = "";
            document.getElementById("edit-imgArchivo").value = "";

            cambiarVista("vista-editar");
        }

        // GUARDAR EDICIÓN
        async function guardarEdicion() {
            if (posicionEnLaLista === null) return;

            let titulo = document.getElementById("edit-tit").value.trim();
            let texto = document.getElementById("edit-txt").value.trim();
            let categoria = document.getElementById("edit-categoria").value;

            if (titulo === "" || texto === "") {
                alert("¡El título y el contenido son obligatorios!");
                return;
            }

            try {
                let imagen = await obtenerImagen(
                    "edit-imgArchivo",
                    "edit-imgUrl",
                    document.getElementById("edit-imgActual").value
                );

                miListaDePublicaciones[posicionEnLaLista] = {
                    titulo: titulo,
                    categoria: categoria,
                    imagen: imagen,
                    texto: texto
                };

                posicionEnLaLista = null;
                cambiarVista("vista-publicaciones");
            } catch (error) {
                alert("No se pudo cargar la imagen.");
            }
        }

        // MOSTRAR LA PANTALLA INICIAL
        cambiarVista("vista-inicio");
    </script>
