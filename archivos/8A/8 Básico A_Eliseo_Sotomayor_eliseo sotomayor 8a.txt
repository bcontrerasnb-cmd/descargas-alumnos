    <section id="vista-crear">
        <h2>Nueva Publicación</h2>

        <input type="hidden" id="edit-imgActual">

    <!-- (EL CEREBRO DEL BLOG VERSIÓN) -->
    <script>
        // ==========================================
        // 1. LA LISTA DE COMPRAS (El Arreglo)
        // ==========================================
        
        //Guardamos toda la información en una Lista ordenadita (Arreglo). Empieza vacía [].
        let miListaDePublicaciones = [];
        
        // Esta es una memoria para recordar en qué "número de fila" de la lista está el post que queremos editar.
        let posicionEnLaLista = null;

        // ==========================================
        // 2. NAVEGACIÓN Y AYUDANTES
        // ==========================================
        
        // TAREA: CAMBIAR ENTRE PANTALLAS
        function cambiarVista(pantallaDestino) {
            // Escondemos todas las pantallas (display: none).
            document.getElementById('vista-inicio').style.display = 'none';
            document.getElementById('vista-publicaciones').style.display = 'none';
            document.getElementById('vista-crear').style.display = 'none';
            document.getElementById('vista-editar').style.display = 'none';
            
            // Mostramos solo la pantalla elegida (display: block).
            document.getElementById(pantallaDestino).style.display = 'block';
        }

        // TAREA: PROCESAR LA IMAGEN
        function obtenerImagen(cajaArchivo, cajaLink, cajaFotoVieja = null) {
            let archivoDePC = document.getElementById(cajaArchivo).files[0];
            // .trim() es como una tijera que corta los espacios vacíos que el usuario puso por accidente al principio o al final.
            let linkDeInternet = document.getElementById(cajaLink).value.trim(); 
            
            if (archivoDePC) {
                return URL.createObjectURL(archivoDePC); 
            } else if (linkDeInternet !== "") {
                return linkDeInternet; 
            } else if (cajaFotoVieja) {
                return document.getElementById(cajaFotoVieja).value; 
            }
            return ""; 
        }

        // ==========================================
        // 3. DIBUJAR TODO EL MURO (La magia del Ciclo)
        // ==========================================
        function mostrarPosts() {
            let cajaMuro = document.getElementById("muro");
            
            // PASO CRÍTICO: ¡Borrador mágico! 
            // Borramos toda la pantalla para que no se nos dupliquen las cosas cuando el ciclo empiece a dibujar.
            cajaMuro.innerHTML = "";

            // LA MÁQUINA DE REPETICIÓN (Ciclo For)
            // Esto le dice al computador: "Empieza a contar en 0 (let i = 0). 
            // Repite esta tarea mientras no te pases de la cantidad de cosas que hay en nuestra lista (i < miListaDePublicaciones.length).
            // Y avanza de uno en uno (i++)".
            for (let i = 0; i < miListaDePublicaciones.length; i++) {
                
                // Sacamos la información pura del objeto que está en la fila 'i'.
                let postActual = miListaDePublicaciones[i];

                // Y ahora, le sumamos (+=) a la pantalla el diseño HTML, rellenándolo con la info pura.
                // Fíjate en los botones: Ahora enviamos el número 'i'. Así los botones saben exactamente qué posición de la lista deben borrar o editar.
                cajaMuro.innerHTML += `
                    <div class="tarjeta-post">
                        <h3 class="t-post">${postActual.titulo}</h3>
                        <img src="${postActual.imagen}" class="img-post">
                        <p class="p-post">${postActual.texto}</p>
                        <div class="caja-botones">
                            <button class="btn btn-secundario" onclick="abrirEditar(${i})">Editar</button>
                            <button class="btn btn-peligro" onclick="borrarPost(${i})">Eliminar</button>
                        </div>
                    </div>
                `;
            }
        }

        // ==========================================
        // 4. CREAR NUEVA PUBLICACIÓN
        // ==========================================
        function crearPost() {
            // 1. Rescatamos los textos de las cajas y les cortamos los espacios extra (.trim).
            let tituloEscrito = document.getElementById("tit").value.trim();
            let textoEscrito = document.getElementById("txt").value.trim();
            let imagenFinal = obtenerImagen("imgArchivo", "imgUrl");

            // 2. Si escribieron puras cosas vacías, los frenamos.
            if (tituloEscrito === "" && textoEscrito === "") {
                alert("¡Debes escribir un título o un texto antes de publicar!");
                return;
            }

            // 3.EMPUJAMOS (.push) toda la información empaquetada como un "Objeto" hacia nuestra Lista principal.
            miListaDePublicaciones.push({
                titulo: tituloEscrito,
                imagen: imagenFinal,
                texto: textoEscrito
            });
            
            // 4. Limpiamos las cajas para que el formulario quede listo para el próximo uso.
            document.getElementById("tit").value = "";
            document.getElementById("txt").value = "";
            document.getElementById("imgUrl").value = "";
            document.getElementById("imgArchivo").value = "";

            // 5. Como nuestra Lista acaba de crecer, le ordenamos a nuestra Tarea 3 (mostrarPosts) 
            // que vuelva a encender su Ciclo For y dibuje la lista actualizada en pantalla.
            mostrarPosts();
            
            // 6. Cambiamos de pantalla para ir a ver nuestro muro.
            cambiarVista('vista-publicaciones'); 
        }

        // ==========================================
        // 5. ELIMINAR UNA PUBLICACIÓN
        // ==========================================
        // Esta función recibe un número ('numeroEnLaFila'), que es la posición que el ciclo For le puso al botón.
        function borrarPost(numeroEnLaFila) {
            let confirmacion = confirm("¿Seguro que quieres eliminar esta publicación?");
            
            if (confirmacion === true) {
                // Usamos el poder '.splice()'. Esto se mete directo a nuestra Lista en la memoria,
                // busca la posición exacta 'numeroEnLaFila', destruye '1' elemento, y empuja a los demás hacia arriba para no dejar un hueco.
                miListaDePublicaciones.splice(numeroEnLaFila, 1);
                
                // Como la Lista cambió (tiene una cosa menos), ordenamos al muro que se dibuje de nuevo para actualizarse.
                mostrarPosts();
            }
        }

        // ==========================================
        // 6. EDITAR PUBLICACIONES
        // ==========================================
        
        // Parte 1: Cargar la información. También recibe el número de la fila.
        function abrirEditar(numeroEnLaFila) {
            // 1. Anotamos en nuestro cuaderno global en qué número estamos para que no se nos olvide.
            posicionEnLaLista = numeroEnLaFila;
            
            // 2. Extraemos toda la información desde la memoria de la Lista.
            let postParaCorregir = miListaDePublicaciones[numeroEnLaFila];

            // 3. Rellenamos las cajas de edición con la información pura de la Lista (ya no leemos el HTML).
            document.getElementById("edit-tit").value = postParaCorregir.titulo;
            document.getElementById("edit-txt").value = postParaCorregir.texto;
            document.getElementById("edit-imgActual").value = postParaCorregir.imagen;
            
            // Vaciamos las cajas de imagen nueva.
            document.getElementById("edit-imgArchivo").value = "";
            document.getElementById("edit-imgUrl").value = "";
            
            // Mostramos la pantalla de editar.
            cambiarVista('vista-editar'); 
        }

        // Parte 2: Guardar las correcciones
        function guardarEdicion() {
            // 1. Rescatamos los nuevos textos corregidos.
            let nuevoTitulo = document.getElementById("edit-tit").value.trim();
            let nuevoTexto = document.getElementById("edit-txt").value.trim();
            let nuevaImagen = obtenerImagen("edit-imgArchivo", "edit-imgUrl", "edit-imgActual");
            
            // 2. ¡Sobreescribimos la memoria! Vamos a nuestra Lista, ubicamos la posición exacta
            // que teníamos anotada, y le chancamos la información vieja con la información nueva.
            miListaDePublicaciones[posicionEnLaLista].titulo = nuevoTitulo;
            miListaDePublicaciones[posicionEnLaLista].texto = nuevoTexto;
            miListaDePublicaciones[posicionEnLaLista].imagen = nuevaImagen;
            
            // 3. Como modificamos algo de la Lista, volvemos a dibujar el muro y cambiamos de pantalla.
            mostrarPosts();
            cambiarVista('vista-publicaciones');
        }
    </script>
