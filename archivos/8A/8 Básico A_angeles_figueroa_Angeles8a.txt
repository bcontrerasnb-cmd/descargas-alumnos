<script>

// =====================================
// 1. ARREGLO DE PUBLICACIONES
// =====================================

let miListaDePublicaciones = [
    {
        titulo: "La Selección Chilena de Vóleibol",
        imagen: "",
        texto: "Nuestra selección representa a Chile en competencias nacionales e internacionales. Sus jugadores demuestran esfuerzo, compromiso y pasión por este deporte."
    },
    {
        titulo: "El trabajo en equipo",
        imagen: "",
        texto: "El vóleibol es un deporte colectivo en el que la comunicación, la coordinación y la confianza entre compañeros son fundamentales para conseguir la victoria."
    },
    {
        titulo: "Chile y las competencias internacionales",
        imagen: "",
        texto: "La participación internacional permite a los deportistas chilenos adquirir experiencia, mejorar su nivel y representar con orgullo a nuestro país."
    }
];

let posicionEnLaLista = null;


// =====================================
// 2. CAMBIAR ENTRE PANTALLAS
// =====================================

function cambiarVista(pantallaDestino) {

    document.getElementById('vista-inicio').style.display = 'none';

    document.getElementById('vista-publicaciones').style.display = 'none';

    document.getElementById('vista-crear').style.display = 'none';

    document.getElementById('vista-editar').style.display = 'none';

    document.getElementById(pantallaDestino).style.display = 'block';

    if (pantallaDestino === 'vista-inicio') {
        mostrarInicio();
    }

    if (pantallaDestino === 'vista-publicaciones') {
        mostrarPosts();
    }
}


// =====================================
// 3. OBTENER IMAGEN
// =====================================

function obtenerImagen(cajaArchivo, cajaLink, cajaFotoVieja = null) {

    let archivoDePC =
    document.getElementById(cajaArchivo).files[0];

    let linkDeInternet =
    document.getElementById(cajaLink).value.trim();

    if (archivoDePC) {
        return URL.createObjectURL(archivoDePC);
    }

    else if (linkDeInternet !== "") {
        return linkDeInternet;
    }

    else if (cajaFotoVieja) {
        return document.getElementById(cajaFotoVieja).value;
    }

    return "";
}


// =====================================
// 4. MOSTRAR PUBLICACIONES
// =====================================

function mostrarPosts() {

    let cajaMuro = document.getElementById("muro");

    cajaMuro.innerHTML = "";

    for (let i = 0; i < miListaDePublicaciones.length; i++) {

        let postActual = miListaDePublicaciones[i];

        let tarjeta = document.createElement("article");
        tarjeta.className = "tarjeta-post";

        let titulo = document.createElement("h3");
        titulo.className = "t-post";
        titulo.textContent = postActual.titulo;

        tarjeta.appendChild(titulo);

        if (postActual.imagen !== "") {
            let imagen = document.createElement("img");
            imagen.src = postActual.imagen;
            imagen.className = "img-post";
            imagen.alt = "Imagen de la publicación";

            tarjeta.appendChild(imagen);
        }

        let texto = document.createElement("p");
        texto.className = "p-post";
        texto.textContent = postActual.texto;

        tarjeta.appendChild(texto);

        let botones = document.createElement("div");
        botones.className = "caja-botones";

        let botonEditar = document.createElement("button");
        botonEditar.className = "btn-secundario";
        botonEditar.textContent = "Editar";
        botonEditar.onclick = function() {
            abrirEditar(i);
        };

        let botonEliminar = document.createElement("button");
        botonEliminar.className = "btn-peligro";
        botonEliminar.textContent = "Eliminar";
        botonEliminar.onclick = function() {
            borrarPost(i);
        };

        botones.appendChild(botonEditar);
        botones.appendChild(botonEliminar);

        tarjeta.appendChild(botones);

        cajaMuro.appendChild(tarjeta);
    }
}


// =====================================
// 5. MOSTRAR NOTICIAS EN EL INICIO
// =====================================

function mostrarInicio() {

    let cajaInicio = document.getElementById("inicio-noticias");

    cajaInicio.innerHTML = "";

    for (let i = 0; i < miListaDePublicaciones.length && i < 3; i++) {

        let post = miListaDePublicaciones[i];

        let tarjeta = document.createElement("div");
        tarjeta.className = "tarjeta-post";

        let titulo = document.createElement("h3");
        titulo.className = "t-post";
        titulo.textContent = post.titulo;

        let texto = document.createElement("p");
        texto.className = "p-post";
        texto.textContent = post.texto;

        tarjeta.appendChild(titulo);

        if (post.imagen !== "") {
            let imagen = document.createElement("img");
            imagen.src = post.imagen;
            imagen.className = "img-post";
            imagen.alt = "Fotografía de la noticia";
            tarjeta.appendChild(imagen);
        }

        tarjeta.appendChild(texto);
        cajaInicio.appendChild(tarjeta);
    }
}


// =====================================
// 6. CREAR PUBLICACION
// =====================================

function crearPost() {

    let tituloEscrito =
    document.getElementById("tit").value.trim();

    let textoEscrito =
    document.getElementById("txt").value.trim();

    let imagenFinal =
    obtenerImagen("imgArchivo", "imgUrl");

    if (tituloEscrito === "" && textoEscrito === "") {

        alert("¡Debes escribir un título o un texto!");
        return;
    }

    miListaDePublicaciones.push({

        titulo: tituloEscrito,
        imagen: imagenFinal,
        texto: textoEscrito

    });

    document.getElementById("tit").value = "";
    document.getElementById("txt").value = "";
    document.getElementById("imgUrl").value = "";
    document.getElementById("imgArchivo").value = "";

    mostrarPosts();

    cambiarVista('vista-publicaciones');

    alert("¡Tu publicación fue creada!");
}


// =====================================
// 7. ELIMINAR PUBLICACION
// =====================================

function borrarPost(numeroEnLaFila) {

    let confirmacion =
    confirm("¿Seguro que quieres eliminar esta publicación?");

    if (confirmacion === true) {

        miListaDePublicaciones.splice(numeroEnLaFila, 1);

        mostrarPosts();

        alert("Publicación eliminada.");
    }
}


// =====================================
// 8. ABRIR EDICION
// =====================================

function abrirEditar(numeroEnLaFila) {

    posicionEnLaLista = numeroEnLaFila;

    let postParaCorregir =
    miListaDePublicaciones[numeroEnLaFila];

    document.getElementById("edit-tit").value =
    postParaCorregir.titulo;

    document.getElementById("edit-txt").value =
    postParaCorregir.texto;

    document.getElementById("edit-imgActual").value =
    postParaCorregir.imagen;

    document.getElementById("edit-imgArchivo").value = "";

    document.getElementById("edit-imgUrl").value = "";

    cambiarVista('vista-editar');
}


// =====================================
// 9. GUARDAR EDICION
// =====================================

function guardarEdicion() {

    let nuevoTitulo =
    document.getElementById("edit-tit").value.trim();

    let nuevoTexto =
    document.getElementById("edit-txt").value.trim();

    let nuevaImagen =
    obtenerImagen("edit-imgArchivo", "edit-imgUrl", "edit-imgActual");

    if (nuevoTitulo === "" && nuevoTexto === "") {
        alert("Debes escribir un título o contenido.");
        return;
    }

    if (posicionEnLaLista === null ||
        !miListaDePublicaciones[posicionEnLaLista]) {
        alert("No se encontró la publicación.");
        return;
    }

    miListaDePublicaciones[posicionEnLaLista].titulo =
    nuevoTitulo;

    miListaDePublicaciones[posicionEnLaLista].texto =
    nuevoTexto;

    miListaDePublicaciones[posicionEnLaLista].imagen =
    nuevaImagen;

    posicionEnLaLista = null;

    mostrarPosts();

    cambiarVista('vista-publicaciones');

    alert("¡Cambios guardados correctamente!");
}


// =====================================
// 10. INICIAR BLOG
// =====================================

mostrarInicio();

</script>
