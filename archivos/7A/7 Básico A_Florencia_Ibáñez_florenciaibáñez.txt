    <script>
        function cambiarVista(idVistaDestino) {
            document.getElementById('vista-inicio').style.display = 'none';
            document.getElementById('vista-publicaciones').style.display = 'none';
            document.getElementById('vista-crear').style.display = 'none';
            document.getElementById('vista-editar').style.display = 'none';

            document.getElementById(idVistaDestino).style.display = 'block';
        }

        function obtenerImagen(idArchivo, idUrl, idActual = null) {
            let archivo = document.getElementById(idArchivo).files[0];
            let url = document.getElementById(idUrl).value;

            if (archivo) {
                return URL.createObjectURL(archivo);
            } else if (url !== "") {
                return url;
            } else if (idActual) {
                return document.getElementById(idActual).value;
            }

            return "";
        }

        function crearPost() {
            let titulo = document.getElementById("tit").value;
            let texto = document.getElementById("txt").value;
            let imagenFinal = obtenerImagen("imgArchivo", "imgUrl");

            if (titulo === "" && texto === "") {
                alert("¡Debes escribir un título o texto antes de publicar!");
                return;
            }

            document.getElementById("muro").innerHTML += `
                <div class="tarjeta-post">
                    <h3 class="t-post">${titulo}</h3>
                    <img src="${imagenFinal}" class="img-post">
                    <p class="p-post">${texto}</p>
                    <div class="caja-botones">
                        <button class="btn btn-secundario" onclick="abrirEditar(this)">Editar</button>
                        <button class="btn btn-peligro" onclick="borrarPost(this)">Eliminar</button>
                    </div>
                </div>
            `;

            document.getElementById("tit").value = "";
            document.getElementById("txt").value = "";
            document.getElementById("imgUrl").value = "";
            document.getElementById("imgArchivo").value = "";

            cambiarVista('vista-publicaciones');
        }

        function borrarPost(botonApretado) {
            let confirmacion = confirm("¿Seguro que quieres eliminar esta publicación?");

            if (confirmacion === true) {
                botonApretado.parentElement.parentElement.remove();
            }
        }

        let postAEditar = null;

        function abrirEditar(botonApretado) {
            postAEditar = botonApretado.parentElement.parentElement;

            document.getElementById("edit-tit").value = postAEditar.querySelector('.t-post').innerText;
            document.getElementById("edit-txt").value = postAEditar.querySelector('.p-post').innerText;
            document.getElementById("edit-imgActual").value = postAEditar.querySelector('.img-post').src;

            document.getElementById("edit-imgArchivo").value = "";
            document.getElementById("edit-imgUrl").value = "";

            cambiarVista('vista-editar');
        }

        function guardarEdicion() {
            let nuevoTit = document.getElementById("edit-tit").value;
            let nuevoTxt = document.getElementById("edit-txt").value;

            let nuevaImg = obtenerImagen("edit-imgArchivo", "edit-imgUrl", "edit-imgActual");

            postAEditar.querySelector('.t-post').innerText = nuevoTit;
            postAEditar.querySelector('.p-post').innerText = nuevoTxt;
            postAEditar.querySelector('.img-post').src = nuevaImg;

            cambiarVista('vista-publicaciones');
        }
    </script>
