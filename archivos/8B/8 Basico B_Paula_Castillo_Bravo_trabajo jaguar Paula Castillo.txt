    <script>
        // ==========================================
        // 1. LA LISTA DE PUBLICACIONES (Pre-cargada con Jaguares)
        // ==========================================
        let miListaDePublicaciones = [
            {
                titulo: "El Yaguareté: El rey de la selva americana",
                imagen: "https://images.unsplash.com/photo-1549451371-72aa3d30cf0f?auto=format&fit=crop&w=800&q=80",
                texto: "El jaguar (Panthera onca), conocido en Sudamérica como yaguareté, es el felino más grande de América. Su nombre proviene de la lengua guaraní y significa 'el que mata de un salto'. Es un animal solitario, nocturno y un excelente nadador."
            },
            {
                titulo: "Curiosidades asombrosas sobre su mordida",
                imagen: "https://images.unsplash.com/photo-1561731216-c3a4d99437d5?auto=format&fit=crop&w=800&q=80",
                texto: "A diferencia de otros felinos, los jaguares adoran el agua y cruzan ríos con facilidad. Además, poseen la mordida más potente de todos los grandes felinos en relación a su tamaño, capaz de perforar con facilidad caparazones de tortugas y presas acorazadas."
            }
        ];
        
        let posicionEnLaLista = null;

        // ==========================================
        // 2. NAVEGACIÓN Y AYUDANTES
        // ==========================================
        function cambiarVista(pantallaDestino) {
            document.getElementById('vista-inicio').style.display = 'none';
            document.getElementById('vista-publicaciones').style.display = 'none';
            document.getElementById('vista-crear').style.display = 'none';
            document.getElementById('vista-editar').style.display = 'none';
            
            document.getElementById(pantallaDestino).style.display = 'block';
        }

        function obtenerImagen(cajaArchivo, cajaLink, cajaFotoVieja = null) {
            let archivoDePC = document.getElementById(cajaArchivo).files[0];
            let linkDeInternet = document.getElementById(cajaLink).value.trim(); 
            
            if (archivoDePC) {
                return URL.createObjectURL(archivoDePC); 
            } else if (linkDeInternet !== "") {
                return linkDeInternet; 
            } else if (cajaFotoVieja) {
                return document.getElementById(cajaFotoVieja).value; 
            }
            return ""; 
        }

        // ==========================================
        // 3. DIBUJAR TODO EL MURO (La magia del Ciclo)
        // ==========================================
        function mostrarPosts() {
            let cajaMuro = document.getElementById("muro");
            cajaMuro.innerHTML = "";

            if (miListaDePublicaciones.length === 0) {
                cajaMuro.innerHTML = `<p style="text-align: center; color: #6b7280;">Aún no hay publicaciones. ¡Crea la primera!</p>`;
                return;
            }

            for (let i = 0; i < miListaDePublicaciones.length; i++) {
                let postActual = miListaDePublicaciones[i];

                cajaMuro.innerHTML += `
                    <div class="tarjeta-post">
                        <h3 class="t-post">${postActual.titulo}</h3>
                        ${postActual.imagen ? `<img src="${postActual.imagen}" class="img-post">` : ''}
                        <p class="p-post">${postActual.texto}</p>
                        <div class="caja-botones">
                            <button class="btn btn-secundario" onclick="abrirEditar(${i})">Editar</button>
                            <button class="btn btn-peligro" onclick="borrarPost(${i})">Eliminar</button>
                        </div>
                    </div>
                `;
            }
        }

        // ==========================================
        // 4. CREAR NUEVA PUBLICACIÓN
        // ==========================================
        function crearPost() {
            let tituloEscrito = document.getElementById("tit").value.trim();
            let textoEscrito = document.getElementById("txt").value.trim();
            let imagenFinal = obtenerImagen("imgArchivo", "imgUrl");

            if (tituloEscrito === "" && textoEscrito === "") {
                alert("¡Debes escribir un título o un texto antes de publicar!");
                return;
            }

            miListaDePublicaciones.push({
                titulo: tituloEscrito,
                imagen: imagenFinal,
                texto: textoEscrito
            });
            
            document.getElementById("tit").value = "";
            document.getElementById("txt").value = "";
            document.getElementById("imgUrl").value = "";
            document.getElementById("imgArchivo").value = "";

            mostrarPosts();
            cambiarVista('vista-publicaciones'); 
        }

        // ==========================================
        // 5. ELIMINAR UNA PUBLICACIÓN
        // ==========================================
        function borrarPost(numeroEnLaFila) {
            let confirmacion = confirm("¿Seguro que quieres eliminar esta publicación?");
            
            if (confirmacion === true) {
                miListaDePublicaciones.splice(numeroEnLaFila, 1);
                mostrarPosts();
            }
        }

        // ==========================================
        // 6. EDITAR PUBLICACIONES
        // ==========================================
        function abrirEditar(numeroEnLaFila) {
            posicionEnLaLista = numeroEnLaFila;
            let postParaCorregir = miListaDePublicaciones[numeroEnLaFila];

            document.getElementById("edit-tit").value = postParaCorregir.titulo;
            document.getElementById("edit-txt").value = postParaCorregir.texto;
            document.getElementById("edit-imgActual").value = postParaCorregir.imagen;
            
            document.getElementById("edit-imgArchivo").value = "";
            document.getElementById("edit-imgUrl").value = "";
            
            cambiarVista('vista-editar'); 
        }

        function guardarEdicion() {
            let nuevoTitulo = document.getElementById("edit-tit").value.trim();
            let nuevoTexto = document.getElementById("edit-txt").value.trim();
            let nuevaImagen = obtenerImagen("edit-imgArchivo", "edit-imgUrl", "edit-imgActual");
            
            miListaDePublicaciones[posicionEnLaLista].titulo = nuevoTitulo;
            miListaDePublicaciones[posicionEnLaLista].texto = nuevoTexto;
            miListaDePublicaciones[posicionEnLaLista].imagen = nuevaImagen;
            
            mostrarPosts();
            cambiarVista('vista-publicaciones');
        }
    </script>
